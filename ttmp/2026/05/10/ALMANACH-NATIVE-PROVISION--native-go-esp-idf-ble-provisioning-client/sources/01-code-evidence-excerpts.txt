    99	func (c *BLEProvisionCommand) RunIntoGlazeProcessor(ctx context.Context, vals *values.Values, gp middlewares.Processor) error {
   100		s := &BLEProvisionSettings{}
   101		if err := vals.DecodeSectionInto(schema.DefaultSlug, s); err != nil {
   102			return err
   103		}
   104		if s.Action == "provision" && s.SSID == "" {
   105			return errors.New("--ssid is required for action=provision")
   106		}
   107		if s.ServiceName == "" {
   108			return errors.New("--service-name is required; use the value printed by firmware prov_status, e.g. ALM_0F2320")
   109		}
   110		if s.Timeout <= 0 {
   111			return errors.New("--timeout must be greater than zero")
   112		}
   113	
   114		passphrase := s.Passphrase
   115		readPassphrase := false
   116		if s.Action == "provision" && passphrase == "" {
   117			var err error
   118			passphrase, err = readLineFromStdin("WiFi passphrase: ")
   119			if err != nil {
   120				return err
   121			}
   122			readPassphrase = true
   123		}
   124	
   125		pythonPath := s.Python
   126		if pythonPath == "" {
   127			pythonPath = "python3"
   128		}
   129		espProvPath := s.EspProv
   130		if espProvPath == "" {
   131			espProvPath = filepath.Join(s.IDFPath, "tools", "esp_prov", "esp_prov.py")
   132		}
   133	
   134		args, displayArgs, err := buildBLEProvisionPythonArgs(s, espProvPath, passphrase)
   135		if err != nil {
   136			return err
   137		}
   138	
   139		if s.InstallHints {
   140			fmt.Fprintln(os.Stderr, bleProvisionInstallHints(s.IDFPath))
   141		}
   142	
   143		if s.DryRun {
   144			fmt.Fprintf(os.Stderr, "IDF_PATH=%s %s %s\n", s.IDFPath, pythonPath, strings.Join(displayArgs, " "))
   145			return gp.AddRow(ctx, types.NewRow(
   146				types.MRP("action", s.Action),
   147				types.MRP("service_name", s.ServiceName),
   148				types.MRP("ssid", s.SSID),
   149				types.MRP("pop", s.Pop),
   150				types.MRP("idf_path", s.IDFPath),
   151				types.MRP("python", pythonPath),
   152				types.MRP("esp_prov", espProvPath),
   153				types.MRP("command", pythonPath+" "+strings.Join(displayArgs, " ")),
   154				types.MRP("dry_run", true),
   155				types.MRP("read_passphrase_from_stdin", readPassphrase),
   156			))
   157		}
   158	
   159		runCtx, cancel := context.WithTimeout(ctx, time.Duration(s.Timeout)*time.Second)
   160		defer cancel()
   161		cmd := exec.CommandContext(runCtx, pythonPath, args...)
   162		cmd.Env = append(os.Environ(), "IDF_PATH="+s.IDFPath)
   163		cmd.Stdout = os.Stdout
   164		cmd.Stderr = os.Stderr
   165		cmd.Stdin = os.Stdin
   166	
   167		started := time.Now()
   168		err = cmd.Run()
   169		duration := time.Since(started)
   170		exitCode := 0
   171		if err != nil {
   172			var exitErr *exec.ExitError
   173			if errors.As(err, &exitErr) {
   174				exitCode = exitErr.ExitCode()
   175			} else if runCtx.Err() != nil {
   176				exitCode = -1
   177			} else {
   178				return err
   179			}
   180		}
   181	
   182		if addErr := gp.AddRow(ctx, types.NewRow(
   183			types.MRP("action", s.Action),
   184			types.MRP("service_name", s.ServiceName),
   185			types.MRP("ssid", s.SSID),
   186			types.MRP("pop", s.Pop),
   187			types.MRP("idf_path", s.IDFPath),
   188			types.MRP("python", pythonPath),
   189			types.MRP("esp_prov", espProvPath),
   190			types.MRP("command", pythonPath+" "+strings.Join(displayArgs, " ")),
   191			types.MRP("exit_code", exitCode),
   192			types.MRP("duration_ms", duration.Milliseconds()),
   193			types.MRP("read_passphrase_from_stdin", readPassphrase),
   194		)); addErr != nil {
   195			return addErr
   196		}
   197	
   198		if err != nil {
   199			if runCtx.Err() != nil {
   200				return fmt.Errorf("esp_prov.py timed out after %s", duration.Round(time.Second))
   201			}
   202			return fmt.Errorf("esp_prov.py failed with exit code %d", exitCode)
   203		}
   204		return nil
   205	}

   207	func buildBLEProvisionPythonArgs(s *BLEProvisionSettings, espProvPath string, passphrase string) ([]string, []string, error) {
   208		if s.Action == "version" {
   209			protoVer := s.ProtoVer
   210			if protoVer == "" {
   211				protoVer = "v1.1"
   212			}
   213			code := `import asyncio, os, sys
   214	idf = os.environ['IDF_PATH']
   215	sys.path.insert(0, idf + '/components/protocomm/python')
   216	sys.path.insert(1, idf + '/tools/esp_prov')
   217	import esp_prov
   218	async def main():
   219	    tp = await esp_prov.get_transport('ble', sys.argv[1])
   220	    try:
   221	        ok = await esp_prov.version_match(tp, sys.argv[2], True)
   222	        print('==== Verified protocol version successfully ====' if ok else '==== Protocol version mismatch ====')
   223	        raise SystemExit(0 if ok else 1)
   224	    finally:
   225	        await tp.disconnect()
   226	asyncio.run(main())`
   227			args := []string{"-c", code, s.ServiceName, protoVer}
   228			displayArgs := []string{"-c", "<almanach-esp-prov-version-check>", s.ServiceName, protoVer}
   229			return args, displayArgs, nil
   230		}
   231	
   232		args := []string{espProvPath, "--transport", "ble", "--service_name", s.ServiceName, "--sec_ver", fmt.Sprint(s.SecVer)}
   233		if s.Pop != "" {
   234			args = append(args, "--pop", s.Pop)
   235		}
   236		if s.ProtoVer != "" {
   237			args = append(args, "--proto_ver", s.ProtoVer)
   238		}
   239		if s.Verbose {
   240			args = append(args, "--verbose")
   241		}
   242	
   243		switch s.Action {
   244		case "provision":
   245			args = append(args, "--ssid", s.SSID, "--passphrase", passphrase)
   246		case "reset":
   247			args = append(args, "--reset")
   248		case "reprov":
   249			args = append(args, "--reprov")
   250		default:
   251			return nil, nil, fmt.Errorf("unsupported action %q", s.Action)
   252		}
   253	
   254		return args, redactEspProvArgs(args), nil
   255	}

    35	static void make_service_identity(void)
    36	{
    37	    uint8_t mac[6] = {0};
    38	    esp_err_t err = esp_read_mac(mac, ESP_MAC_WIFI_STA);
    39	    if (err != ESP_OK) {
    40	        ESP_LOGW(TAG, "esp_read_mac failed: %s; using fallback identity", esp_err_to_name(err));
    41	        strlcpy(s_service_name, "ALM_SETUP", sizeof(s_service_name));
    42	        strlcpy(s_pop, "alm-setup", sizeof(s_pop));
    43	        return;
    44	    }
    45	
    46	    snprintf(s_service_name, sizeof(s_service_name),
    47	             "ALM_%02X%02X%02X", mac[3], mac[4], mac[5]);
    48	    snprintf(s_pop, sizeof(s_pop),
    49	             "alm-%02x%02x%02x", mac[3], mac[4], mac[5]);

    63	static void provisioning_event_handler(void *arg, esp_event_base_t event_base,
    64	                                       int32_t event_id, void *event_data)
    65	{
    66	    (void)arg;
    67	
    68	    if (event_base == WIFI_PROV_EVENT) {
    69	        switch (event_id) {
    70	        case WIFI_PROV_INIT:
    71	            ESP_LOGI(TAG, "WiFi provisioning manager initialized");
    72	            break;
    73	        case WIFI_PROV_START:
    74	            s_running = true;
    75	            ESP_LOGI(TAG, "BLE WiFi provisioning started");
    76	            break;
    77	        case WIFI_PROV_CRED_RECV: {
    78	            const wifi_sta_config_t *cfg = (const wifi_sta_config_t *)event_data;
    79	            ESP_LOGI(TAG, "Received WiFi credentials for SSID '%s'", (const char *)cfg->ssid);
    80	            break;
    81	        }
    82	        case WIFI_PROV_CRED_FAIL: {
    83	            const wifi_prov_sta_fail_reason_t *reason =
    84	                (const wifi_prov_sta_fail_reason_t *)event_data;
    85	            ESP_LOGE(TAG, "Provisioned WiFi connection failed: %s",
    86	                     (*reason == WIFI_PROV_STA_AUTH_ERROR) ? "authentication failed" : "AP not found");
    87	            break;
    88	        }
    89	        case WIFI_PROV_CRED_SUCCESS:
    90	            ESP_LOGI(TAG, "Provisioned WiFi credentials connected successfully");
    91	            break;
    92	        case WIFI_PROV_END:
    93	            s_running = false;
    94	            ESP_LOGI(TAG, "BLE WiFi provisioning ended");
    95	            wifi_prov_mgr_deinit();
    96	            s_initialized = false;
    97	            break;
    98	        case WIFI_PROV_DEINIT:
    99	            s_running = false;
   100	            s_initialized = false;
   101	            ESP_LOGI(TAG, "WiFi provisioning manager deinitialized");
   102	            break;
   103	        default:
   104	            break;
   105	        }
   106	        return;
   107	    }
   108	
   109	    if (event_base == PROTOCOMM_TRANSPORT_BLE_EVENT) {
   110	        switch (event_id) {
   111	        case PROTOCOMM_TRANSPORT_BLE_CONNECTED:
   112	            s_client_connected = true;
   113	            ESP_LOGI(TAG, "BLE provisioning client connected");
   114	            break;
   115	        case PROTOCOMM_TRANSPORT_BLE_DISCONNECTED:
   116	            s_client_connected = false;
   117	            ESP_LOGI(TAG, "BLE provisioning client disconnected");
   118	            break;
   119	        default:
   120	            break;
   121	        }
   122	        return;
   123	    }
   124	
   125	    if (event_base == PROTOCOMM_SECURITY_SESSION_EVENT) {
   126	        switch (event_id) {
   127	        case PROTOCOMM_SECURITY_SESSION_SETUP_OK:
   128	            s_security_ok = true;
   129	            ESP_LOGI(TAG, "Provisioning security session established");
   130	            break;
   131	        case PROTOCOMM_SECURITY_SESSION_CREDENTIALS_MISMATCH:
   132	            s_security_ok = false;
   133	            ESP_LOGE(TAG, "Provisioning PoP mismatch");
   134	            break;
   135	        case PROTOCOMM_SECURITY_SESSION_INVALID_SECURITY_PARAMS:
   136	            s_security_ok = false;
   137	            ESP_LOGE(TAG, "Provisioning security parameters invalid");
   138	            break;
   139	        default:
   140	            break;
   141	        }
   142	    }
   143	}

   167	esp_err_t provisioning_mgr_init(void)
   168	{
   169	    if (s_initialized) {
   170	        return ESP_OK;
   171	    }
   172	
   173	    make_service_identity();
   174	
   175	    wifi_prov_mgr_config_t config = {
   176	        .scheme = wifi_prov_scheme_ble,
   177	        .scheme_event_handler = WIFI_PROV_SCHEME_BLE_EVENT_HANDLER_FREE_BTDM,
   178	        .app_event_handler = WIFI_PROV_EVENT_HANDLER_NONE,
   179	    };
   180	
   181	    ESP_RETURN_ON_ERROR(wifi_prov_mgr_init(config), TAG, "wifi_prov_mgr_init");
   182	
   183	    wifi_prov_scheme_ble_set_service_uuid(s_custom_service_uuid);
   184	
   185	    ESP_RETURN_ON_ERROR(register_event_handlers(), TAG, "register provisioning handlers");
   186	
   187	    s_initialized = true;
   188	    s_running = false;
   189	    s_client_connected = false;
   190	    s_security_ok = false;
   191	    return ESP_OK;
   192	}

   195	{
   196	    if (out_provisioned == NULL) {
   197	        return ESP_ERR_INVALID_ARG;
   198	    }
   199	    ESP_RETURN_ON_ERROR(provisioning_mgr_init(), TAG, "ensure provisioning manager initialized");
   200	    return wifi_prov_mgr_is_provisioned(out_provisioned);

   203	esp_err_t provisioning_mgr_start_if_needed(bool *out_started)
   204	{
   205	    if (out_started != NULL) {
   206	        *out_started = false;
   207	    }
   208	
   209	    ESP_RETURN_ON_ERROR(provisioning_mgr_init(), TAG, "ensure provisioning manager initialized");
   210	
   211	    if (s_running) {
   212	        ESP_LOGI(TAG, "BLE provisioning already running");
   213	        return ESP_OK;
   214	    }
   215	
   216	    bool provisioned = false;
   217	    ESP_RETURN_ON_ERROR(wifi_prov_mgr_is_provisioned(&provisioned), TAG, "check provisioned state");
   218	    if (provisioned) {
   219	        ESP_LOGI(TAG, "Device already provisioned; BLE provisioning not started");
   220	        return ESP_OK;
   221	    }
   222	
   223	    ESP_RETURN_ON_ERROR(provisioning_mgr_start_force(), TAG, "start provisioning");
   224	    if (out_started != NULL) {
   225	        *out_started = true;
   226	    }
   227	    return ESP_OK;
   228	}

    74	async function discoverEndpointCharacteristics(provisioningService, log) {
    75	  const characteristics = await provisioningService.getCharacteristics();
    76	  const endpoints = new Map();
    77	  log(`Discovered ${characteristics.length} provisioning characteristic(s)`);
    78	
    79	  for (const characteristic of characteristics) {
    80	    let endpointName = null;
    81	    try {
    82	      const descriptors = await characteristic.getDescriptors();
    83	      for (const descriptor of descriptors) {
    84	        if (!descriptor.uuid.toLowerCase().includes("2901")) continue;
    85	        const value = await descriptor.readValue();
    86	        endpointName = dataViewToText(value).trim().toLowerCase();
    87	        break;
    88	      }
    89	    } catch (error) {
    90	      log(`Could not read descriptor for ${characteristic.uuid}: ${error.message || error}`);
    91	    }
    92	
    93	    if (endpointName) {
    94	      endpoints.set(endpointName, characteristic);
    95	      log(`Mapped endpoint ${endpointName} -> ${characteristic.uuid}`);
    96	    }
    97	  }
    98	
    99	  for (const [name, uuid] of Object.entries(FALLBACK_ENDPOINT_UUIDS)) {
   100	    if (endpoints.has(name)) continue;
   101	    const characteristic = characteristics.find((candidate) => candidate.uuid.toLowerCase() === uuid);
   102	    if (characteristic) {
   103	      endpoints.set(name, characteristic);
   104	      log(`Mapped endpoint ${name} by fallback UUID ${uuid}`);
   105	    }
   106	  }
   107	
   108	  return endpoints;

   207	async def scan_wifi_APs(sel_transport, tp, sec):
   208	    APs = []
   209	    group_channels = 0
   210	    readlen = 100
   211	    if sel_transport in ['softap', 'httpd']:
   212	        # In case of softAP/httpd we must perform the scan on individual channels, one by one,
   213	        # so that the Wi-Fi controller gets ample time to send out beacons (necessary to
   214	        # maintain connectivity with authenticated stations. As scanning one channel at a
   215	        # time will be slow, we can group more than one channels to be scanned in quick
   216	        # succession, hence speeding up the scan process. Though if too many channels are
   217	        # present in a group, the controller may again miss out on sending beacons. Hence,
   218	        # the application must should use an optimum value. The following value usually
   219	        # works out in most cases
   220	        group_channels = 5
   221	    elif sel_transport == 'ble':
   222	        # Read at most 4 entries at a time. This is because if we are using BLE transport
   223	        # then the response packet size should not exceed the present limit of 256 bytes of
   224	        # characteristic value imposed by protocomm_ble. This limit may be removed in the
   225	        # future
   226	        readlen = 4
   227	    try:
   228	        message = prov.scan_start_request(sec, blocking=True, group_channels=group_channels)
   229	        start_time = time.time()
   230	        response = await tp.send_data('prov-scan', message)
   231	        stop_time = time.time()
   232	        print('++++ Scan process executed in ' + str(stop_time - start_time) + ' sec')
   233	        prov.scan_start_response(sec, response)
   234	
   235	        message = prov.scan_status_request(sec)
   236	        response = await tp.send_data('prov-scan', message)
   237	        result = prov.scan_status_response(sec, response)
   238	        print('++++ Scan results : ' + str(result['count']))
   239	        if result['count'] != 0:
   240	            index = 0
   241	            remaining = result['count']
   242	            while remaining:
   243	                count = [remaining, readlen][remaining > readlen]
   244	                message = prov.scan_result_request(sec, index, count)
   245	                response = await tp.send_data('prov-scan', message)
   246	                APs += prov.scan_result_response(sec, response)
   247	                remaining -= count
   248	                index += count
   249	
   250	    except RuntimeError as e:
   251	        on_except(e)
   252	        return None
   253	
   254	    return APs

    38	    def security1_session(self, response_data):
    39	        # protocomm security1 FSM which interprets/forms
    40	        # protobuf packets according to present state of session
    41	        if (self.session_state == security_state.REQUEST1):
    42	            self.session_state = security_state.RESPONSE1_REQUEST2
    43	            return self.setup0_request()
    44	        elif (self.session_state == security_state.RESPONSE1_REQUEST2):
    45	            self.session_state = security_state.RESPONSE2
    46	            self.setup0_response(response_data)
    47	            return self.setup1_request()
    48	        elif (self.session_state == security_state.RESPONSE2):

    50	            self.setup1_response(response_data)

    67	    def setup0_request(self):
    68	        # Form SessionCmd0 request packet using client public key
    69	        setup_req = proto.session_pb2.SessionData()
    70	        setup_req.sec_ver = proto.session_pb2.SecScheme1
    71	        self.__generate_key()
    72	        setup_req.sec1.sc0.client_pubkey = self.client_public_key
    73	        self._print_verbose(f'Client Public Key:\t0x{self.client_public_key.hex()}')
    74	        return setup_req.SerializeToString().decode('latin-1')

    76	    def setup0_response(self, response_data):
    77	        # Interpret SessionResp0 response packet
    78	        setup_resp = proto.session_pb2.SessionData()
    79	        setup_resp.ParseFromString(str_to_bytes(response_data))
    80	        self._print_verbose('Security version:\t' + str(setup_resp.sec_ver))
    81	        if setup_resp.sec_ver != proto.session_pb2.SecScheme1:
    82	            raise RuntimeError('Incorrect security scheme')
    83	
    84	        self.device_public_key = setup_resp.sec1.sr0.device_pubkey
    85	        # Device random is the initialization vector
    86	        device_random = setup_resp.sec1.sr0.device_random
    87	        self._print_verbose(f'Device Public Key:\t0x{self.device_public_key.hex()}')
    88	        self._print_verbose(f'Device Random:\t0x{device_random.hex()}')
    89	
    90	        # Calculate Curve25519 shared key using Client private key and Device public key
    91	        sharedK = self.client_private_key.exchange(X25519PublicKey.from_public_bytes(self.device_public_key))
    92	        self._print_verbose(f'Shared Key:\t0x{sharedK.hex()}')

    94	        # If PoP is provided, XOR SHA256 of PoP with the previously
    95	        # calculated Shared Key to form the actual Shared Key
    96	        if len(self.pop) > 0:
    97	            # Calculate SHA256 of PoP
    98	            h = hashes.Hash(hashes.SHA256(), backend=default_backend())
    99	            h.update(self.pop)
   100	            digest = h.finalize()
   101	            # XOR with and update Shared Key
   102	            sharedK = a_xor_b(sharedK, digest)
   103	            self._print_verbose(f'Updated Shared Key (Shared key XORed with PoP):\t0x{sharedK.hex()}')
   104	        # Initialize the encryption engine with Shared Key and initialization vector
   105	        cipher = Cipher(algorithms.AES(sharedK), modes.CTR(device_random), backend=default_backend())
   106	        self.cipher = cipher.encryptor()
   107	
   108	    def setup1_request(self):
   109	        # Form SessionCmd1 request packet using encrypted device public key
   110	        setup_req = proto.session_pb2.SessionData()
   111	        setup_req.sec_ver = proto.session_pb2.SecScheme1
   112	        setup_req.sec1.msg = proto.sec1_pb2.Session_Command1
   113	        # Encrypt device public key and attach to the request packet
   114	        client_verify = self.cipher.update(self.device_public_key)
   115	        self._print_verbose(f'Client Proof:\t0x{client_verify.hex()}')
   116	        setup_req.sec1.sc1.client_verify_data = client_verify
   117	        return setup_req.SerializeToString().decode('latin-1')
   118	
   119	    def setup1_response(self, response_data):
   120	        # Interpret SessionResp1 response packet
   121	        setup_resp = proto.session_pb2.SessionData()
   122	        setup_resp.ParseFromString(str_to_bytes(response_data))
   123	        # Ensure security scheme matches
   124	        if setup_resp.sec_ver == proto.session_pb2.SecScheme1:
   125	            # Read encrypyed device verify string
   126	            device_verify = setup_resp.sec1.sr1.device_verify_data
   127	            self._print_verbose(f'Device Proof:\t0x{device_verify.hex()}')
   128	            # Decrypt the device verify string
   129	            enc_client_pubkey = self.cipher.update(setup_resp.sec1.sr1.device_verify_data)
   130	            # Match decryped string with client public key
   131	            if enc_client_pubkey != self.client_public_key:
   132	                raise RuntimeError('Failed to verify device!')
   133	        else:
   134	            raise RuntimeError('Unsupported security protocol')

   137	        return self.cipher.update(data)
